cmake_minimum_required(VERSION 3.16)
project(ProjectHorizon VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HORIZON_SDL "Build the optional SDL2 pixel-art window" ON)
option(HORIZON_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(horizon_core
    src/MissionProgression.cpp
    src/RobotEyes.cpp
    src/Types.cpp src/Inventory.cpp src/WorldGrid.cpp src/Unit7Entity.cpp
    src/NASA_ISRU_Refinery.cpp src/RocketRepairSystem.cpp src/ClimaxEngine.cpp
    src/RescueMission.cpp src/Game.cpp src/PhobosClock.cpp src/DangerSystem.cpp src/Presentation.cpp src/SoundEffects.cpp src/Renderer.cpp src/ShipInteriorRenderer.cpp src/Minimap.cpp src/Preview.cpp src/SpaceBackground.cpp src/TargetTool.cpp src/IsometricScene.cpp src/IsometricDemo.cpp src/PlanetTerrain.cpp src/SceneLighting.cpp src/SceneShadows.cpp src/PlanetProjection.cpp)
target_include_directories(horizon_core PUBLIC include)
target_compile_options(horizon_core PRIVATE -Wall -Wextra -Wpedantic)
if(HORIZON_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(horizon_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(horizon_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(horizon src/main.cpp)
target_link_libraries(horizon PRIVATE horizon_core)
if(HORIZON_SDL)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
    endif()
    if(TARGET PkgConfig::SDL2)
        target_sources(horizon PRIVATE src/SDLRenderer.cpp)
        target_compile_definitions(horizon PRIVATE HORIZON_HAS_SDL=1)
        target_link_libraries(horizon PRIVATE PkgConfig::SDL2)
    else()
        message(STATUS "SDL2 unavailable: terminal, demo, and PPM renderer remain available")
    endif()
endif()
enable_testing()
add_executable(horizon_tests tests/test_main.cpp)
target_link_libraries(horizon_tests PRIVATE horizon_core)
add_test(NAME horizon_systems COMMAND horizon_tests)
add_test(NAME horizon_mission COMMAND horizon --demo)
add_executable(horizon_battery_tests tests/test_battery.cpp)
target_link_libraries(horizon_battery_tests PRIVATE horizon_core)
add_test(NAME horizon_battery COMMAND horizon_battery_tests)
add_executable(horizon_environment_tests tests/test_environment.cpp)
target_link_libraries(horizon_environment_tests PRIVATE horizon_core)
add_test(NAME horizon_environment COMMAND horizon_environment_tests)

add_executable(horizon_mission1_tests tests/test_mission1.cpp)
target_link_libraries(horizon_mission1_tests PRIVATE horizon_core)
add_test(NAME horizon_mission1 COMMAND horizon_mission1_tests)
add_executable(horizon_ship_tests tests/test_ship.cpp)
target_link_libraries(horizon_ship_tests PRIVATE horizon_core)
add_test(NAME horizon_ship COMMAND horizon_ship_tests)
add_executable(horizon_rescue_tests tests/test_rescue.cpp)
target_link_libraries(horizon_rescue_tests PRIVATE horizon_core)
add_test(NAME horizon_rescue COMMAND horizon_rescue_tests)
